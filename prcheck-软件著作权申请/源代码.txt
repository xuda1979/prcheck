    added_nonbreaking = any(
        not c.breaking and c.kind in ("endpoint", "parameter", "schema")
        for c in changes
    )
    suggestion = "patch"
    if breaking:
        suggestion = "major"
    elif added_nonbreaking:
        suggestion = "minor"
    return {
        "from": old_version,
        "to": new_version,
        "breaking_changes": breaking,
        "changes_count": len(changes),
        "suggested_bump": suggestion,
    }
def diff_spec_text(old_text: str, new_text: str) -> tuple[list[dict], dict]:
    old = load_spec(old_text, "old")
    new = load_spec(new_text, "new")
    changes = compare_specs(old, new)
    semver = infer_semver(old.version_string, new.version_string, changes)
    return [c.to_dict() for c in changes], semver

def _retry_after(req_hdrs, resp_headers) -> float | None:
    sentinel = object()
    ra = resp_headers.get("Retry-After", sentinel)
    if ra is not sentinel and ra:
        try:
            return max(0.0, float(ra))
        except (TypeError, ValueError):
            pass
    for headers in (req_hdrs, resp_headers):
        reset = headers.get("X-RateLimit-Reset", sentinel)
        if reset is sentinel or not reset:
            continue
        try:
            return max(0.0, float(reset) - time.time())
        except (TypeError, ValueError):
            continue
    return None

def verify_signature(secret: str, body: bytes, signature_header: str) -> bool:
    if not signature_header:

    rank = SEVERITY_ORDER.get(min_severity, 0)

def filter_issues(
    issues: list[Issue],
    *,
    security_only: bool = False,
    data_only: bool = False,
    behavior_only: bool = False,
    min_severity: str = "info",
) -> list[Issue]:
    """Apply category + minimum-severity filters.

    This is the single filter entry point shared by the CLI, the GitHub Action
    and the webhook bridge, so all entry points honor identical filtering.
    """
    rank = SEVERITY_ORDER.get(min_severity, 0)
    out: list[Issue] = []
    for i in issues:
        if security_only and i.category != "security":
            continue
        if data_only and i.category != "data":
            continue
        if behavior_only and i.category != "behavior":
            continue
        if SEVERITY_ORDER.get(i.severity, 0) < rank:
            continue
        out.append(i)
    return out

def _pat(pattern: str) -> re.Pattern:
    if pattern not in _PATTERN_CACHE:

def _should_skip_path(path: str) -> bool:
    """Never flag lockfiles or vendored data files."""
    base = path.replace("\\", "/").split("/")[-1]
    if base in ("Pipfile.lock", "poetry.lock", "yarn.lock", "package-lock.json",
                "Cargo.lock", "go.sum", "composer.lock", "Gemfile.lock", "pnpm-lock.yaml"):
        return True
    if path.endswith((".min.js", ".min.css", ".map", ".svg", ".png", ".jpg", ".ico")):
        return True
    return False

def _strip_path(raw: str) -> str:
    """Turn ``b/src/x.py`` into ``src/x.py``; handle ``/dev/null``."""
    raw = raw.strip()
    if raw == "/dev/null":
        return ""
    # git mangles paths with spaces as `b/a b/file`; just strip the prefix.
    if raw.startswith("a/") or raw.startswith("b/"):
        return raw[2:]
    return raw

def _compare_schema_attrs(old: dict, new: dict) -> list[tuple[str, Any, Any]]:
    """Yield (propertyOrKey, oldVal, newVal) for changed primitive props.

    ``properties``/``required`` are reported at property level below, so they
    are skipped here to avoid duplicating (and mis-classifying) the whole-map
    comparison as a flat attribute change.

    out = []
    for k in set(old) | set(new):
        if k in ("description", "example", "default", "properties", "required"):
            continue
        if k in old and k in new and not _deep_equals(old[k], new[k]):
            out.append((k, old[k], new[k]))
    return out

def _attr_breaking(attr: str, old_v: Any, new_v: Any) -> bool:
    if attr in ("type", "required", "enum"):
        return True
    if attr == "additionalProperties":
        return bool(old_v) and not bool(new_v)
    return False


def _prop_breaking(old_prop: Any, new_prop: Any) -> bool:
    """A property-level change is breaking when an old value becomes invalid."""
    if not isinstance(old_prop, dict) or not isinstance(new_prop, dict):
        return old_prop != new_prop and old_prop is not None
    if "type" in old_prop and "type" in new_prop:
        return old_prop["type"] != new_prop["type"]
    if "$ref" in old_prop and "$ref" in new_prop:
        return old_prop["$ref"] != new_prop["$ref"]
    # an unknown, non-primitive structural change is not classified as
    # breaking by default (conservative)
    return False

def verify_signature(secret: str, body: bytes, signature_header: str) -> bool:
    """Constant-time check of a GitHub ``X-Hub-Signature-256`` header.

    ``signature_header`` is the raw header value, e.g.
    ``sha256=9a7f...``. Returns False when the header is missing, malformed,
    or does not match the HMAC-SHA256 of ``body`` under ``secret``.
    """
    if not signature_header:
        return False
    try:
        scheme, _, expected = signature_header.partition("=")
        if scheme != "sha256" or not expected:
            return False
        actual = hmac.new(secret.encode("utf-8"), body, hashlib.sha256).hexdigest()
        return hmac.compare_digest(actual, expected)
    except (TypeError, ValueError):
        return False
